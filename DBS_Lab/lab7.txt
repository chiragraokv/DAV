-- 2. Write a PL/SQL block to display the letter grade(0-4: F; 4-5: E; 5-6: D; 6-7: C;              7-8: B; 8-9: A; 9-10: A+} of given student.

DECLARE
roll StudentTable.RollNo%TYPE := &roll_no;
gpa StudentTable.GPA%TYPE;
grade VARCHAR(2);
BEGIN
SELECT GPA into gpa FROM StudentTable WHERE RollNo = roll;
IF gpa < 4 THEN grade := 'F';
ELSIF gpa < 5 THEN grade := 'E';
ELSIF gpa < 6 THEN grade := 'D';
ELSIF gpa < 7 THEN grade := 'C';
ELSIF gpa < 8 THEN grade := 'B';
ELSIF gpa < 9 THEN grade := 'A';
ELSIF gpa < 10 THEN grade := 'A+';
END IF;
DBMS_OUTPUT.PUT_LINE('Grade of Roll no. ' || roll || ': ' || grade);
END;

-- 3. Input the date of issue and date of return for a book. Calculate and display the fine 
with the appropriate message using a PL/SQL block.

DECLARE
issue_date DATE := TO_DATE('&issue_date', 'DD-MM-YYYY');
return_date DATE := TO_DATE('&return_date', 'DD-MM-YYYY');
days_late NUMBER;
fine NUMBER := 0;
BEGIN
days_late := return_date - issue_date;
IF days_late <= 7 THEN fine := 0;
ELSIF days_late <= 15 THEN fine := (days_late - 7);
ELSIF days_late <= 30 THEN fine := 8 + (days_late - 15) * 2;
ELSE fine := 5.00;
END IF;

DBMS_OUTPUT.PUT_LINE('Fine: Rs. ' || fine);
END;


-- 4. Write a PL/SQL block to print the letter grade of all the students(RollNo: 1 - 5).

DECLARE
counter NUMBER := 1;
gpa StudentTable.GPA%TYPE;
grade VARCHAR2(2);
BEGIN
LOOP
SELECT GPA INTO gpa FROM StudentTable WHERE RollNo = counter;
IF gpa < 4 THEN grade := 'F';
ELSIF gpa < 5 THEN grade := 'E';
ELSIF gpa < 6 THEN grade := 'D';
ELSIF gpa < 7 THEN grade := 'C';
ELSIF gpa < 8 THEN grade := 'B';
ELSIF gpa < 9 THEN grade := 'A';
ELSIF gpa < 10 THEN grade := 'A+';
END IF;
DBMS_OUTPUT.PUT_LINE('Grade of Roll no. ' || counter || ': ' || grade);
counter := counter + 1;
EXIT WHEN counter > 5;
END LOOP;
END;

-- 6. Write a PL/SQL block to find the student with max. GPA without using aggregate function. 
DECLARE
max_gpa StudentTable.GPA%TYPE := 0;
best_student StudentTable.RollNo%TYPE;
BEGIN
FOR rec IN (SELECT RollNo, GPA FROM StudentTable) LOOP
IF rec.GPA > max_gpa THEN
max_gpa := rec.GPA;
best_student := rec.RollNo;
END IF;
END LOOP;
DBMS_OUTPUT.PUT_LINE('Max GPA: ' || max_gpa || ' | Roll No: ' || best_student);
END;

);
