-- 1. The HRD manager has decided to raise the salary of all the Instructors in a given department number by 5%. Whenever, any such raise is given to the instructor, a record for the same is maintained in the salary_raise table. It includes the Instuctor Id, the date when the raise was given and the actual raise amount. Write a PL/SQL block to update the salary of each Instructor and insert a record in the salary_raise table.

DECLARE
    v_dept_name     instructor.dept_name%TYPE := 'Computer Science'; 
    v_id            instructor.ID%TYPE;
    v_salary        instructor.salary%TYPE;
    v_raise_amt     NUMBER(8,2);

    CURSOR inst_cursor IS
        SELECT ID, salary
        FROM instructor
        WHERE dept_name = v_dept_name;

BEGIN
    OPEN inst_cursor;
    IF inst_cursor%ISOPEN THEN
        DBMS_OUTPUT.PUT_LINE('Cursor opened successfully.');
    END IF;
    LOOP
        FETCH inst_cursor INTO v_id, v_salary;
        EXIT WHEN inst_cursor%NOTFOUND;
        v_raise_amt := v_salary * 0.05;
        UPDATE instructor
        SET salary = salary + v_raise_amt
        WHERE ID = v_id;
        INSERT INTO salary_raise (Instructor_Id, Raise_date, Raise_amt)
        VALUES (v_id, SYSDATE, v_raise_amt);
        DBMS_OUTPUT.PUT_LINE('Processed Instructor ID: ' || v_id ||
                             ' | Rows Processed: ' || inst_cursor%ROWCOUNT);
        IF inst_cursor%FOUND THEN
            DBMS_OUTPUT.PUT_LINE('Record found and updated.');
        END IF;
    END LOOP;
    CLOSE inst_cursor;
    DBMS_OUTPUT.PUT_LINE('Total Instructors Updated: ' || inst_cursor%ROWCOUNT);

    COMMIT;
END;
/

-- 2. Write a PL/SQL block that will display the ID, name, dept_name and tot_cred of the first 10 students with lowest total credit.

DECLARE
    CURSOR lowest_credit_cursor IS
        SELECT ID, name, dept_name, tot_cred
        FROM student
        ORDER BY tot_cred ASC
        FETCH FIRST 10 ROWS ONLY;
    
    student_record lowest_credit_cursor%ROWTYPE;
BEGIN
    OPEN lowest_credit_cursor;
    
    LOOP
        FETCH lowest_credit_cursor INTO student_record;
        EXIT WHEN lowest_credit_cursor%NOTFOUND;
        
        DBMS_OUTPUT.PUT_LINE('ID: ' || student_record.ID || ', Name: ' || student_record.name ||
                             ', Dept: ' || student_record.dept_name || ', Total Credits: ' || student_record.tot_cred);
    END LOOP;
    
    CLOSE lowest_credit_cursor;
END;

-- 3. Print the Course details and the total number of students registered for each course along with the course details - (Course-id, title, dept-name, credits, instructor_name, building, room-number, time-slot-id, tot_student_no )

DECLARE
    CURSOR course_cursor IS
        SELECT c.course_id, c.title, c.dept_name, c.credits, i.name AS instructor_name, 
               s.building, s.room_number, s.time_slot_id, COUNT(t.ID) AS tot_student_no
        FROM course c
        JOIN section s ON c.course_id = s.course_id
        JOIN teaches t ON s.course_id = t.course_id AND s.sec_id = t.sec_id
        JOIN instructor i ON t.ID = i.ID
        LEFT JOIN takes ta ON s.course_id = ta.course_id AND s.sec_id = ta.sec_id
        GROUP BY c.course_id, c.title, c.dept_name, c.credits, i.name, s.building, s.room_number, s.time_slot_id;
    
    course_record course_cursor%ROWTYPE;
BEGIN
    OPEN course_cursor;
    
    LOOP
        FETCH course_cursor INTO course_record;
        EXIT WHEN course_cursor%NOTFOUND;
        
        DBMS_OUTPUT.PUT_LINE('Course ID: ' || course_record.course_id || ', Title: ' || course_record.title ||
                             ', Dept: ' || course_record.dept_name || ', Credits: ' || course_record.credits ||
                             ', Instructor: ' || course_record.instructor_name || ', Building: ' || course_record.building ||
                             ', Room: ' || course_record.room_number || ', Time Slot: ' || course_record.time_slot_id ||
                             ', Total Students: ' || course_record.tot_student_no);
    END LOOP;
    
    CLOSE course_cursor;
END;

-- 4. Find all students who take the course with Course-id: CS101 and if he/ she has less than 30 total credit (tot-cred), deregister the student from that course. (Delete the entry in Takes table)

DECLARE
    CURSOR student_cursor IS
        SELECT t.ID
        FROM takes t
        JOIN course c ON t.course_id = c.course_id
        JOIN student s ON t.ID = s.ID
        WHERE c.course_id = 'CS101' AND s.tot_cred < 30;
BEGIN
    FOR student_record IN student_cursor LOOP
        DELETE FROM takes
        WHERE ID = student_record.ID AND course_id = 'CS101';
    END LOOP;
    
    COMMIT;
END;

-- 5. Alter StudentTable(refer Lab No. 8 Exercise) by resetting column LetterGrade to F. Then write a PL/SQL block to update the table by mapping GPA to the corresponding letter grade for each student.

DECLARE
    CURSOR student_cursor IS
        SELECT ID, gpa FROM student;
    
    student_record student_cursor%ROWTYPE;
BEGIN
    UPDATE student
    SET LetterGrade = 'F';
    
    FOR student_record IN student_cursor LOOP
        IF student_record.gpa >= 9 THEN
            UPDATE student SET LetterGrade = 'A' WHERE ID = student_record.ID;
        ELSIF student_record.gpa >= 8 THEN
            UPDATE student SET LetterGrade = 'B' WHERE ID = student_record.ID;
        ELSIF student_record.gpa >= 7 THEN
            UPDATE student SET LetterGrade = 'C' WHERE ID = student_record.ID;
        ELSE
            UPDATE student SET LetterGrade = 'D' WHERE ID = student_record.ID;
        END IF;
    END LOOP;
    
    COMMIT;
END;

-- 6. Write a PL/SQL block to print the list of Instructors teaching a specified course.

DECLARE
    v_course_id VARCHAR2(8) := 'CS-101';
    CURSOR instructor_cursor IS
        SELECT i.name
        FROM instructor i
        JOIN teaches t ON i.ID = t.ID
        JOIN section s ON t.course_id = s.course_id
        WHERE s.course_id = v_course_id;
    
    instructor_record instructor_cursor%ROWTYPE;
BEGIN
    OPEN instructor_cursor;
    
    LOOP
        FETCH instructor_cursor INTO instructor_record;
        EXIT WHEN instructor_cursor%NOTFOUND;
        
        DBMS_OUTPUT.PUT_LINE('Instructor: ' || instructor_record.name);
    END LOOP;
    
    CLOSE instructor_cursor;
END;

-- 7. Write a PL/SQL block to list the students who have registered for a course taught by his/her advisor.

DECLARE
    CURSOR advisor_student_cursor IS
        SELECT s.ID, s.name
        FROM student s
        JOIN advisor a ON s.ID = a.s_ID
        JOIN instructor i ON a.i_ID = i.ID
        JOIN teaches t ON i.ID = t.ID
        JOIN section sec ON t.course_id = sec.course_id
        WHERE s.ID = a.s_ID AND sec.course_id = t.course_id;
    
    student_record advisor_student_cursor%ROWTYPE;
BEGIN
    OPEN advisor_student_cursor;
    
    LOOP
        FETCH advisor_student_cursor INTO student_record;
        EXIT WHEN advisor_student_cursor%NOTFOUND;
        
        DBMS_OUTPUT.PUT_LINE('Student ID: ' || student_record.ID || ', Student Name: ' || student_record.name);
    END LOOP;
    
    CLOSE advisor_student_cursor;
END;
