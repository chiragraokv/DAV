-- 2. Based on the University Database Schema in Lab 2, write a procedure which takes the dept_name as input parameter and lists all the instructors associated with the department as well as list all the courses offered by the department. Also, write an anonymous block with the procedure call.

CREATE OR REPLACE PROCEDURE dept_info(p_dept_name IN VARCHAR2) IS
BEGIN
    DBMS_OUTPUT.PUT_LINE('Department: ' || p_dept_name);
    DBMS_OUTPUT.PUT_LINE('Instructors:');
    FOR instr IN (
        SELECT name, ID, salary
        FROM instructor
        WHERE dept_name = p_dept_name
    ) LOOP
        DBMS_OUTPUT.PUT_LINE('ID: ' || instr.ID || ', Name: ' || instr.name || ', Salary: ' || instr.salary);
    END LOOP;

    DBMS_OUTPUT.PUT_LINE('');

    DBMS_OUTPUT.PUT_LINE('Courses:');
    FOR crs IN (
        SELECT course_id, title, credits
        FROM course
        WHERE dept_name = p_dept_name
    ) LOOP
        DBMS_OUTPUT.PUT_LINE('Course ID: ' || crs.course_id || ', Title: ' || crs.title || ', Credits: ' || crs.credits);
    END LOOP;
END;
/

BEGIN
    dept_info('Comp. Sci.');
END;
/

-- 3. Based on the University Database Schema in Lab 2, write a Pl/Sql block of code that lists the most popular course (highest number of students take it) for each of the departments. It should make use of a procedure course_popular which finds the most popular course in the given department.

CREATE OR REPLACE PROCEDURE course_popular(p_dept_name IN VARCHAR2) IS
    v_course_id      course.course_id%TYPE;
    v_student_count  NUMBER;
BEGIN
    SELECT c.course_id, COUNT(t.ID) AS student_count
    INTO v_course_id, v_student_count
    FROM course c
    LEFT JOIN takes t ON c.course_id = t.course_id
    WHERE c.dept_name = p_dept_name
    GROUP BY c.course_id
    ORDER BY COUNT(t.ID) DESC
    FETCH FIRST 1 ROWS ONLY;

    DBMS_OUTPUT.PUT_LINE('Department: ' || p_dept_name);
    DBMS_OUTPUT.PUT_LINE('Most Popular Course: ' || v_course_id);
END;
/

BEGIN
FOR dept IN (SELECT dept_name from department) LOOP
   course_popular(dept.dept_name);
END LOOP;
END;
/

-- 4. Based on the University Database Schema in Lab 2, write a procedure which takes the dept-name as input parameter and lists all the students associated with the department as well as list all the courses offered by the department. Also, write an anonymous block with the procedure call.

        WHERE dept_name = p_dept_name

        DBMS_OUTPUT.PUT_LINE('Course ID: ' || crs.course_id || ', Title: ' || crs.title);
    END LOOP;
END;

-- 6. Based on the University Database Schema in Lab 2, write a Pl/Sql block of code that lists the highest paid Instructor in each of the Department. It should make use of a function department_highest which returns the highest paid Instructor for the given branch.

CREATE OR REPLACE FUNCTION department_highest(p_dept_name IN VARCHAR2)
RETURN VARCHAR2
IS
    v_name instructor.name%TYPE;
BEGIN
    SELECT name INTO v_name FROM instructor
    WHERE dept_name = p_dept_name
    ORDER BY salary DESC
    FETCH FIRST 1 ROWS ONLY;
    RETURN v_name;
END;
/

BEGIN
    FOR dept_rec IN (SELECT dept_name FROM department) LOOP
        DBMS_OUTPUT.PUT_LINE('Department: ' || dept_rec.dept_name || ' | Highest Paid Instructor: ' || department_highest(dept_rec.dept_name)
        );
    END LOOP;
END;
/

-- 7. Based on the University Database Schema in Lab 2, create a package to include the following:
a)
A named procedure to list the instructor_names of given department
64
b)
A function which returns the max salary for the given department
c)
Write a PL/SQL block to demonstrate the usage of above package components

CREATE OR REPLACE PACKAGE university_pkg AS
    PROCEDURE list_instructors(p_dept_name IN VARCHAR2);
    FUNCTION get_max_salary(p_dept_name IN VARCHAR2) RETURN NUMBER;
END university_pkg;
/

CREATE OR REPLACE PACKAGE BODY university_pkg AS
    PROCEDURE list_instructors(p_dept_name IN VARCHAR2) IS
    BEGIN
        FOR rec IN (
            SELECT name FROM instructor WHERE dept_name = p_dept_name
        ) LOOP
            DBMS_OUTPUT.PUT_LINE(rec.name);
        END LOOP;
    END list_instructors;

    FUNCTION get_max_salary(p_dept_name IN VARCHAR2) RETURN NUMBER IS
        v_max_salary NUMBER;
    BEGIN
        SELECT MAX(salary) INTO v_max_salary FROM instructor
        WHERE dept_name = p_dept_name;
        RETURN v_max_salary;
    END get_max_salary;

END university_pkg;
/

DECLARE
    v_max_salary NUMBER;
BEGIN
    university_pkg.list_instructors('Comp. Sci.');
    v_max_salary := university_pkg.get_max_salary('Comp. Sci.');
    DBMS_OUTPUT.PUT_LINE('Maximum Salary in Comp. Sci. Department: ' || v_max_salary);
END;
/

-- 8. Write a PL/SQL procedure to return simple and compound interest (OUT parameters) along with the Total Sum (IN OUT) i.e. Sum of Principle and Interest taking as input the principle, rate of interest and number of years (IN parameters). Call this procedure from an anonymous block.

CREATE OR REPLACE PROCEDURE calculate_interest(
    p_principal IN NUMBER,
    p_rate      IN NUMBER,
    p_years     IN NUMBER,
    p_simple    OUT NUMBER,
    p_compound  OUT NUMBER,
    p_total     IN OUT NUMBER
) IS
BEGIN
    p_simple   := p_principal * p_rate * p_years / 100;
    p_compound := p_principal * (POWER(1 + p_rate/100, p_years) - 1);
    p_total    := p_principal + p_compound;
END calculate_interest;
/

DECLARE
    v_principal NUMBER := 10000;
    v_rate      NUMBER := 5;
    v_years     NUMBER := 3;
    v_simple    NUMBER;
    v_compound  NUMBER;
    v_total     NUMBER := 0;
BEGIN
    calculate_interest(v_principal, v_rate, v_years, v_simple, v_compound, v_total);    
    DBMS_OUTPUT.PUT_LINE('SI: ' || v_simple || ', CI: ' || v_compound || ', Total: ' || v_total);
END;
/
